案例 1：51 单片机智能小车

功能说明

  • L298N 驱动双路直流电机，实现前进 / 后退 / 左右转向 / 停止
  • 可通过按键切换运行模式（自动循迹 / 手动遥控）
  • 预留红外循迹、超声波避障扩展接口

硬件清单

  • STC89C52RC 最小系统板 ×1
  • L298N 电机驱动模块 ×1、直流减速电机 ×2、万向轮 ×1
  • 锂电池组（7.4V）或 4 节 18650 供电
  • 按键、LED、车架与连接线若干

核心代码片段

/* 51 智能小车：L298N 双电机控制 */
#include <reg52.h>

sbit IN1 = P1^0;   /* 左电机 A */
sbit IN2 = P1^1;
sbit IN3 = P1^2;   /* 右电机 B */
sbit IN4 = P1^3;

void delay_ms(unsigned int ms);   /* 软件延时，见完整工程 */

void motor_forward(void){ IN1=1; IN2=0; IN3=1; IN4=0; }
void motor_back(void)   { IN1=0; IN2=1; IN3=0; IN4=1; }
void motor_left(void)   { IN1=1; IN2=0; IN3=0; IN4=0; }
void motor_right(void)  { IN1=0; IN2=0; IN3=1; IN4=0; }
void motor_stop(void)   { IN1=0; IN2=0; IN3=0; IN4=0; }

void main(void){
    while(1){
        motor_forward(); delay_ms(2000);
        motor_left();    delay_ms(800);
        motor_right();   delay_ms(800);
        motor_stop();    delay_ms(1000);
    }
}

案例 2：温湿度采集系统（LCD1602）

功能说明

  • DHT11 单总线协议读取温湿度，刷新周期 2 秒
  • LCD1602 第一行显示温度、第二行显示湿度
  • 超阈值时 LED 报警（可在程序里修改上下限）

硬件清单

  • STC89C52RC 最小系统板 ×1
  • DHT11 温湿度传感器模块 ×1
  • LCD1602 液晶 + IIC 转接板（或直驱 11 线）
  • 10kΩ 上拉电阻、LED、按键

核心代码片段

/* DHT11 单总线读取一字节（示例） */
#include <reg52.h>
sbit DQ = P2^0;          /* 数据线 */

void delay_us(unsigned int t);   /* 微秒级延时 */

unsigned char dht11_read_byte(void){
    unsigned char i, dat = 0;
    for(i = 0; i < 8; i++){
        while(DQ == 0);          /* 等待 50us 低电平结束 */
        delay_us(40);            /* 约 40us 后采样 */
        dat <<= 1;
        if(DQ == 1) dat |= 0x01; /* 高电平为 1 */
        while(DQ == 1);          /* 等待高电平结束 */
    }
    return dat;
}
/* 主程序：启动信号 → 读湿度整/小、温度整/小、校验和 → LCD1602 显示 */

案例 3：STM32 OLED 数字电压表

功能说明

  • ADC1 通道采样 0~3.3V 模拟电压，12 位分辨率
  • OLED（SSD1306，I2C）实时显示当前电压，两位小数
  • 按键切换量程：直接采样 / 电阻分压（0~30V）

硬件清单

  • STM32F103C8T6 最小系统板 ×1
  • 0.96 寸 OLED（I2C 接口）×1
  • 10kΩ 电位器 ×1（信号源）、电阻分压网络
  • 按键 ×2、杜邦线若干

核心代码片段

/* STM32 HAL：ADC 采集 + OLED 显示电压 */
#include "main.h"
#include "stdio.h"
#include "oled.h"          /* SSD1306 软件 I2C 驱动 */

uint16_t adc_value;
float voltage;

void Voltage_Display(void){
    char buf[20];

    HAL_ADC_Start(&hadc1);                       /* 启动转换 */
    if(HAL_ADC_PollForConversion(&hadc1, 100) == HAL_OK){
        adc_value = HAL_ADC_GetValue(&hadc1);    /* 0~4095 */
        voltage = adc_value * 3.3f / 4096.0f;    /* 换算电压 */

        sprintf(buf, "V=%.2fV", voltage);
        OLED_ShowString(0, 2, buf);
    }
}

案例 4：ESP32 手机 WiFi 远程控制 LED

功能说明

  • STA 模式连接家庭 WiFi，启动内置 Web 服务器
  • 手机浏览器访问 IP 即显示控制页面，远程开关 LED
  • 串口打印连接状态与访问日志，便于调试

硬件清单

  • ESP32 DevKitC 开发板 ×1
  • LED + 220Ω 限流电阻（或使用板载 LED）
  • 5V / 1A 以上 USB 供电
  • 可扩展：继电器模块控制 220V 家电

核心代码片段

/* ESP32 Arduino：手机浏览器远程控制 LED（引脚 2） */
#include <WiFi.h>

const char* ssid     = "你的WiFi名称";
const char* password = "你的WiFi密码";
WiFiServer server(80);

void setup(){
    pinMode(2, OUTPUT);
    Serial.begin(115200);

    WiFi.begin(ssid, password);
    while(WiFi.status() != WL_CONNECTED){ delay(500); }
    Serial.print("IP: "); Serial.println(WiFi.localIP());
    server.begin();
}

void loop(){
    WiFiClient client = server.available();
    if(!client) return;
    String req = "";
    while(client.available()) req += (char)client.read();

    if(req.indexOf("GET /led/on")  >= 0) digitalWrite(2, HIGH);
    if(req.indexOf("GET /led/off") >= 0) digitalWrite(2, LOW);

    client.println("HTTP/1.1 200 OK");
    client.println("Content-Type: text/html");
    client.println();
    client.println("<a href='/led/on'>开灯</a><br><a href='/led/off'>关灯</a>");
    delay(1);
    client.stop();
}
